Bowyer-Watson算法详解

算法步骤简述：

  1. 初始化：创建一个巨大的三角形包含所有点
  2. 逐点插入：每次加入一个新点
  3. 找坏三角形：找到外接圆包含新点的三角形
  4. 删除坏三角形：把这些三角形删掉
  5. 重新连接：用新点连接空洞的边界
  6. 清理：最后删除初始的大三角形

public class BowyerWatson { public List<Triangle> Triangulate(List<Point> points) { // 步骤1: 创建超级三角形 var superTriangle = CreateSuperTriangle(points); var triangles = new List<Triangle> { superTriangle }; // 步骤2: 逐点插入 foreach (var point in points) { // 找到"坏"三角形（外接圆包含新点的三角形） var badTriangles = new List<Triangle>(); foreach (var triangle in triangles) { if (IsPointInCircumcircle(point, triangle)) { badTriangles.Add(triangle); } } // 找到多边形边界（坏三角形围成的空洞边界） var polygon = FindPolygonBoundary(badTriangles); // 删除坏三角形 foreach (var badTriangle in badTriangles) { triangles.Remove(badTriangle); } // 用新点连接多边形边界，形成新三角形 foreach (var edge in polygon) { var newTriangle = new Triangle(point, edge.P1, edge.P2); triangles.Add(newTriangle); } } // 步骤3: 删除包含超级三角形顶点的三角形 return RemoveSuperTriangles(triangles, superTriangle); } }

关键函数详解：

// 判断点是否在三角形的外接圆内 private bool IsPointInCircumcircle(Point point, Triangle triangle) { var circumcenter = triangle.GetCircumcenter(); var radius = Distance(circumcenter, triangle.A); var distanceToPoint = Distance(circumcenter, point); return distanceToPoint < radius; // 距离小于半径就在圆内 } // 找到多边形边界（这是最复杂的部分） private List<Edge> FindPolygonBoundary(List<Triangle> badTriangles) { var edges = new List<Edge>(); // 收集所有坏三角形的边 foreach (var triangle in badTriangles) { edges.Add(new Edge(triangle.A, triangle.B)); edges.Add(new Edge(triangle.B, triangle.C)); edges.Add(new Edge(triangle.C, triangle.A)); } // 只保留出现奇数次的边（边界边） var boundary = new List<Edge>(); foreach (var edge in edges) { int count = edges.Count(e => e.Equals(edge) || e.Equals(edge.Reverse())); if (count == 1) // 只出现一次的边就是边界 { boundary.Add(edge); } } return boundary; }
